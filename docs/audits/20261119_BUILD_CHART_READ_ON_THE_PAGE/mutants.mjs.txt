const results = [];
const vitest = path.join(TREE, 'node_modules', 'vitest', 'vitest.mjs');
for (const [id, file, from, to, tests] of M) {
  if (ONLY.size && !ONLY.has(id)) continue;
  const abs = path.join(TREE, file);
  const orig = readFileSync(abs, 'utf8');
  const n = orig.split(from).length - 1;
  if (n !== 1) { results.push({ id, verdict: n === 0 ? 'SKIPPED (anchor missing)' : `AMBIGUOUS (${n})` }); console.log(id, results.at(-1).verdict); continue; }
  writeFileSync(abs, orig.replace(from, () => to));
  const out = path.join(TREE, `mut-${id}.json`);
  if (existsSync(out)) rmSync(out);
  let code = 0;
  try {
    execFileSync(process.execPath, [vitest, 'run', ...tests, '--reporter=json', `--outputFile=${out}`, '--no-cache'], { cwd: TREE, stdio: 'pipe', timeout: 600000 });
  } catch (e) { code = e.status ?? 1; }
  writeFileSync(abs, orig);
  let verdict;
  try {
    const j = JSON.parse(readFileSync(out, 'utf8'));
    const loadFail = j.testResults.filter((r) => r.status === 'failed' && (r.assertionResults || []).length === 0).length;
    if (loadFail) verdict = `BROKEN (${loadFail} file(s) failed to load)`;
    else verdict = j.numFailedTests > 0 ? `KILLED (${j.numFailedTests} failed)` : (code ? `? exit ${code}, 0 failed` : 'SURVIVED');
  } catch { verdict = `BROKEN (no report, exit ${code})`; }
  results.push({ id, verdict });
  console.log(id, verdict);
}
writeFileSync(path.join(TREE, 'mutants-report.json'), JSON.stringify(results, null, 1));
const tally = results.reduce((t, r) => { const k = r.verdict.split(' ')[0]; t[k] = (t[k] || 0) + 1; return t; }, {});
console.log(JSON.stringify(tally));
