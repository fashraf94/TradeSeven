// ───────────────────────────── the flag-off differential (SCRATCH, never committed) ─────────────────────────────
// Runs in two trees (origin/main and the build's head) with CHART_READ_MODE 'off' and writes one JSON trace; the two
// traces must be identical. Per step: the DOM (React useId renumbered), every chart-library call, every request, focus.
import { writeFileSync } from 'node:fs';
const renumber = (html) => { const ids = new Map(); return html.replace(/_r_[0-9a-z]+_|:r[0-9a-z]+:/g, (m) => { if (!ids.has(m)) ids.set(m, `ID${ids.size}`); return ids.get(m); }); };
const focusOf = () => { const a = document.activeElement; if (!a || a === document.body) return 'body'; return `${a.tagName}${[...a.attributes].filter((x) => x.name.startsWith('data-') || x.name === 'aria-label').map((x) => `[${x.name}=${x.value}]`).join('')}`; };
const libLog = () => JSON.stringify([rec.series, rec.lines, rec.markers].map((l) => l.map((x) => { try { return JSON.parse(JSON.stringify(x)); } catch { return String(x); } })));
const steps = [];
let lastNet = 0;
let lastLib = '';
const snap = (name) => {
  const lib = libLog();
  steps.push({ name, html: renumber(document.body.innerHTML), lib: lib === lastLib ? 'same' : lib, net: JSON.stringify(net.calls.slice(lastNet)), focus: focusOf() });
  lastLib = lib; lastNet = net.calls.length;
};
const btn = (t) => $$('button').find((b) => b.textContent.trim() === t);
const keyDown = async (el, k) => { await act(async () => { el.dispatchEvent(new KeyboardEvent('keydown', { key: k, bubbles: true, cancelable: true })); }); await flush(); };
const tryClick = async (el, name) => { if (el) await click(el); snap(name + (el ? '' : ' (absent)')); };

    await tryClick(pill('chart'), 'browse: Chart');
    await tryClick(btn('1W'), 'browse: 1W');
    await tryClick(btn('SMA'), 'browse: SMA');
    await tryClick(btn('S/R'), 'browse: S/R');
    await tryClick(btn('1D'), 'browse: 1D');
    await tryClick(pill('analysis'), 'browse: Analysis');

    await tryClick(pill('chart'), 'battle: Chart');

    await tryClick(pill('analysis'), 'battle: Analysis');
    await tryClick(pill('bomb'), 'battle: 💣');
    await tryClick(pill('chart'), 'battle: Chart from 💣');
    await keyDown(document.body, 'Escape');
    snap('battle: Escape');
    writeFileSync(process.env.OFF_DIFF_OUT, JSON.stringify(steps));
