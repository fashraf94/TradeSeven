// 6 Fundamental Investor — crystal. Icosahedral facets, inset for seams, locking top-back to bottom-front.
let ICO = null;
function icosa() {
  if (ICO) return ICO;
  const p = (1 + Math.sqrt(5)) / 2, verts = [[0, 1, p], [0, -1, p], [0, 1, -p], [0, -1, -p], [1, p, 0], [-1, p, 0], [1, -p, 0], [-1, -p, 0], [p, 0, 1], [-p, 0, 1], [p, 0, -1], [-p, 0, -1]].map(v => V3(...v).normalize());
  const adj = (a, b) => Math.abs(verts[a].distanceTo(verts[b]) - 1.0515) < .02, faces = [];
  for (let a = 0; a < 12; a++) for (let b = a + 1; b < 12; b++) for (let c = b + 1; c < 12; c++) if (adj(a, b) && adj(b, c) && adj(a, c)) { const f = [a, b, c], n = verts[b].clone().sub(verts[a]).cross(verts[c].clone().sub(verts[a])); if (n.dot(verts[a]) < 0) { f[1] = c; f[2] = b; } faces.push(f); }
  const cen = f => verts[f[0]].clone().add(verts[f[1]]).add(verts[f[2]]).multiplyScalar(1 / 3), seed = V3(-.3, .9, -.3).normalize(), score = f => cen(f).dot(seed);
  let si = 0, best = -9; faces.forEach((f, i) => { const s = score(f); if (s > best) { best = s; si = i; } });
  const order = [faces[si]], used = new Set([si]); let qi = 0;
  while (order.length < faces.length && qi < order.length) { const f = order[qi++]; faces.map((g, i) => ({ g, i })).filter(o => !used.has(o.i) && f.filter(v => o.g.includes(v)).length === 2).sort((a, b) => score(b.g) - score(a.g)).forEach(o => { used.add(o.i); order.push(o.g); }); }
  ICO = { verts, faces: order }; return ICO;
}
function buildFund(E) {

    const vs = f.map(i => I.verts[i].clone()), c = vs[0].clone().add(vs[1]).add(vs[2]).multiplyScalar(1 / 3), dir = c.clone().normalize(), pts = vs.map(v => v.clone().lerp(c, .06)), etch = [];
    for (let e = 0; e < A.etch; e++) etch.push(polyFlat(pts.map(v => v.clone().lerp(c, .3 + .3 * e).addScaledVector(dir, .005))));
    const p = makePlate(polyPrism(pts, dir.clone().negate(), .07), E.mat, etch, { order: fi / cnt, thresh: 12 }); p.userData.dir.copy(dir); p.userData.fi = fi; setAway(p, 1.2, 1, r); plates.push(p); g.add(p);
  });
  function update(t, S, pal, fit) {

    for (const p of plates) { const u = p.userData; applyFit(p, fitK(fit, u.order, .9)); p.position.addScaledVector(u.dir, .09 * op - .05 * stress + .1 * S.lift * Math.max(0, Math.sin(t * 2.6 - u.fi * .5))); if (S.lift) u.edgeK *= .55 + .45 * Math.max(0, Math.sin(t * 3 - u.fi * .55)); }
  }

}

  if (kind !== 'none' && BUILDERS[kind]) { F.armor = BUILDERS[kind]({ stage: armor, A: ARMOR[armor] || ARMOR[0], seed, mat, brainR: (CORE[core] || CORE[0]).R }); group.add(F.armor.group); }
  FORMS.set(key, F); return F;
}

const BL = new Map();
function bloom(ctx, size, res, strength) {
  const bs = Math.max(16, Math.round(size / 4)); let b = BL.get(bs); if (!b) { const A = document.createElement('canvas'), B = document.createElement('canvas'); A.width = A.height = B.width = B.height = bs; b = { A, B, a: A.getContext('2d'), b: B.getContext('2d') }; BL.set(bs, b); }
  b.a.clearRect(0, 0, bs, bs); b.a.drawImage(renderer.domElement, 0, GL - res, res, res, 0, 0, bs, bs);
  b.b.clearRect(0, 0, bs, bs); b.b.filter = `blur(${Math.max(1.5, bs * .07)}px)`; b.b.drawImage(b.A, 0, 0); b.b.filter = 'none';
  ctx.globalCompositeOperation = 'lighter'; ctx.globalAlpha = strength; ctx.drawImage(b.B, 0, 0, size, size); ctx.globalAlpha = 1; ctx.globalCompositeOperation = 'source-over';
}
const camDir = new THREE.Vector3(), invQ = new THREE.Quaternion();
function render(ctx, spec, t) {
  const S = STATES[spec.state] || STATES.idle, pal = PAL[spec.pal] || PAL.teal, size = spec.size, small = size < 90, dpr = Math.min(window.devicePixelRatio || 1, small ? 3 : 2), res = clamp(Math.round(size * dpr), 48, GL), F = getForm(spec), fit = spec.fit == null ? 1 : spec.fit, A = ARMOR[spec.armor | 0] || ARMOR[0];
  F.group.rotation.set(0, .16 * Math.sin(t * .27 + (spec.seed | 0)) + S.scan * .3 * Math.sin(t * .9) + (spec.turn || 0), -.14 * S.lean);
  F.group.position.set(0, S.lift * .07 * Math.max(0, Math.sin(t * 2.6)), 0); F.group.scale.setScalar(F.armor ? F.armor.frame : 1.25);
  const D = 4.7 * (spec.zoom || 1), yaw = .55, pitch = .25; camera.position.set(D * Math.sin(yaw) * Math.cos(pitch), D * Math.sin(pitch), D * Math.cos(yaw) * Math.cos(pitch)); camera.lookAt(0, 0, 0); camera.updateMatrixWorld();
  scene.fog.near = D - .7; scene.fog.far = D + 3.1;
  camDir.copy(camera.position).normalize().applyQuaternion(invQ.copy(F.group.quaternion).invert());
  if (F.brain) F.brain.update(t, S, pal, camDir);
  setFitMode(spec.unfit ? 'off' : 'on');
  if (F.armor) { F.armor.update(t, S, pal, fit); const eb = .95 * A.edge * S.bright * (small ? 1.4 : 1) * (spec.brain === false ? .8 : 1); for (const p of F.armor.plates) { const u = p.userData; u.em.color.set(pal.near); u.em.opacity = clamp(eb * u.edgeK, 0, 1); u.em2.color.set(pal.near); u.em2.opacity = clamp(.42 * eb * u.edgeK, 0, 1); } }
  F.mat.userData.rim.value.set(pal.near); F.mat.userData.rimK.value = .6 * A.rim * S.bright * (small ? 1.5 : 1) * (spec.brain === false ? .6 : 1) * (.6 + .4 * fit);
  if (CUR !== F.group) { if (CUR) scene.remove(CUR); scene.add(F.group); CUR = F.group; }
  renderer.setScissorTest(true); renderer.setViewport(0, 0, res, res); renderer.setScissor(0, 0, res, res); renderer.clear(true, true, false); renderer.render(scene, camera);
  ctx.clearRect(0, 0, size, size); ctx.save(); if (spec.face < 0) { ctx.translate(size, 0); ctx.scale(-1, 1); }
  ctx.drawImage(renderer.domElement, 0, GL - res, res, res, 0, 0, size, size);
  if (!small && size >= 160 && !window.AF_NOBLOOM) bloom(ctx, size, res, clamp(.65 * S.bright * (S.bloom > 1 ? 1.3 : 1), 0, 1));
  ctx.restore();
}

// ── registry: one rAF loop with a frame budget; offscreen canvases skipped; stills when motion is off ──
const REG = new Map(); let raf = 0, IO = null, cursor = 0, frameNo = 0, nextId = 0; const T0 = performance.now();
window.AF_MOTION = window.AF_MOTION == null ? !(window.matchMedia && window.matchMedia('(prefers-reduced-motion: reduce)').matches) : window.AF_MOTION;
const stillT = spec => ((META[spec.kind] || META.none).still || 5) + (spec.core | 0) * .7 + (spec.armor | 0) * .4 + (spec.seed | 0) * .31 + (spec.state === 'celebrating' ? .6 : 0);
function draw(it, t) { try { render(it.ctx, it.spec, t); } catch (e) { if (!it.err) { it.err = 1; console.error('AF3 render failed', JSON.stringify(it.spec), String(e && (e.stack || e.message || e))); } } }
function frame(now) {
  raf = 0; if (!window.AF_MOTION || document.hidden) return; frameNo++;
  const T = (now - T0) / 1000, items = []; REG.forEach(it => { if (it.vis && it.px >= 60) items.push(it); }); const n = items.length;
  if (n) { const start = performance.now(); let k = 0; for (; k < n; k++) { const it = items[(cursor + k) % n]; if (it.spec.size < 100 && (frameNo + it.id) % 3) continue; draw(it, it.t0 + T); if (performance.now() - start > 5) { k++; break; } } cursor = (cursor + k) % n; }
  raf = requestAnimationFrame(frame);
}
function kick() { if (window.AF_MOTION && !raf && !document.hidden) raf = requestAnimationFrame(frame); }
function ensureIO() { if (IO || !('IntersectionObserver' in window)) return; IO = new IntersectionObserver(es => { es.forEach(e => { const it = REG.get(e.target); if (it) { it.vis = e.isIntersecting; it.px = e.boundingClientRect.width; } }); kick(); }, { threshold: 0 }); }
function mount(canvas, spec) {
  ensureIO(); const dpr = Math.min(window.devicePixelRatio || 1, spec.size < 100 ? 3 : 2); canvas.width = Math.round(spec.size * dpr); canvas.height = canvas.width;
  const ctx = canvas.getContext('2d'); ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
  const it = { canvas, ctx, spec, id: nextId++, t0: ((spec.seed | 0) * 1.37 + (spec.core | 0) * 2.1 + (spec.armor | 0) * .9) % 10, vis: true, px: spec.size };
  REG.set(canvas, it); if (IO) IO.observe(canvas);
  draw(it, window.AF_MOTION ? it.t0 + (performance.now() - T0) / 1000 : stillT(spec)); kick(); return it;
}
function unmount(canvas) { const it = REG.get(canvas); if (!it) return; REG.delete(canvas); if (IO) IO.unobserve(canvas); }
function setMotion(on) { window.AF_MOTION = !!on; if (on) kick(); else REG.forEach(it => draw(it, stillT(it.spec))); window.dispatchEvent(new Event('af-motion')); }
document.addEventListener('visibilitychange', kick);
window.AF3 = { mount, unmount, setMotion, META, KINDS, STATES, PAL, CORE, ARMOR, getForm, renderer };
window.dispatchEvent(new Event('af3-ready'));
