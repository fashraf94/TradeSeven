function useConnector(stageRef, originRef, glyphRef, deps) {
  const [path, setPath] = React.useState(null);
  React.useLayoutEffect(() => {
    const calc = () => { const S = stageRef.current, F = originRef.current, G = glyphRef.current; if (!S || !F || !G) { setPath(null); return; } const sr = S.getBoundingClientRect(), fr = F.getBoundingClientRect(), gr = G.getBoundingClientRect(), k = (sr.width / S.offsetWidth) || 1; const f = { x: (fr.left - sr.left) / k, y: (fr.top - sr.top) / k, w: fr.width / k }, g = { x: (gr.left - sr.left) / k, y: (gr.top - sr.top) / k, h: gr.height / k }, fcx = f.x + f.w / 2, fcy = f.y + f.w / 2, R = f.w / 2.6, ex = g.x - 2, ey = g.y + g.h / 2, th = Math.atan2(ey - fcy, ex - fcx), sx = fcx + Math.cos(th) * R, sy = fcy + Math.sin(th) * R, mx = (sx + ex) / 2; setPath({ d: `M${sx.toFixed(1)} ${sy.toFixed(1)} C${mx.toFixed(1)} ${sy.toFixed(1)}, ${mx.toFixed(1)} ${ey.toFixed(1)}, ${ex.toFixed(1)} ${ey.toFixed(1)}`, sx, sy }); };
    calc(); const id = requestAnimationFrame(calc); if (document.fonts && document.fonts.ready) document.fonts.ready.then(calc); return () => cancelAnimationFrame(id);
  }, deps);
  return path;
}
