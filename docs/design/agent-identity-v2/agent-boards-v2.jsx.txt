function useMotion() {
  const [m, setM] = React.useState(!!window.AF_MOTION);
  React.useEffect(() => { const f = () => setM(!!window.AF_MOTION); window.addEventListener('af-motion', f); return () => window.removeEventListener('af-motion', f); }, []);
  return m;
}
// One agent view. kind: 'none' (bare Shadow) or an armor set. core/armor: growth stages 0..2. fit: equip progress 0..1. brain=false renders the empty armor.
function Agent({ kind = 'none', core = 0, armor = 0, state = 'idle', pal = 'teal', size = 200, fit = 1, brain = true, face = 1, zoom = 1, unfit = false, seed = 1, label, style }) {
  const ref = React.useRef(null);
  React.useEffect(() => {
    const c = ref.current; let live = true, mounted = false;
    const go = () => { if (!live || mounted) return; mounted = true; window.AF3.mount(c, { kind, core, armor, state, pal, size, fit, brain, face, zoom, unfit, seed }); };
    if (window.AF3) go(); else window.addEventListener('af3-ready', go, { once: true });
    return () => { live = false; window.removeEventListener('af3-ready', go); if (mounted) window.AF3.unmount(c); };
  }, [kind, core, armor, state, pal, size, fit, brain, face, zoom, unfit, seed]);
  return <canvas ref={ref} role="img" aria-label={label || `${kind === 'none' ? 'Shadow, unarmored' : 'Shadow wearing ' + NAMES[kind]} — ${state}`} style={{ width: size, height: size, display: 'block', flexShrink: 0, ...style }} />;
}
function Board({ n, title, note, children, style }) {
  return <section data-screen-label={`Board ${n} · ${title}`} aria-label={`Board ${n}: ${title}`} style={{ width: 1720, background: FF.bg, border: `1px solid ${FF.hair2}`, borderRadius: 24, padding: '28px 32px 32px', color: FF.ink, fontFamily: SANS, ...style }}>
    <header style={{ display: 'flex', alignItems: 'baseline', gap: 18, marginBottom: 22, flexWrap: 'wrap' }}><FfEyebrow color={FF.teal}>Board {n}</FfEyebrow><h2 style={{ margin: 0, fontSize: 24, fontWeight: 700, letterSpacing: '-0.015em' }}>{title}</h2>{note && <span style={{ fontSize: 13.5, color: FF.ink2, textWrap: 'pretty' }}>{note}</span>}</header>
    {children}
  </section>;
}
function Panel({ children, style, pad = 18 }) { return <div style={{ background: FF.surface, border: `1px solid ${FF.hair}`, borderRadius: 18, padding: pad, minWidth: 0, ...style }}>{children}</div>; }
function Motion({ children, style }) { return <div style={{ marginTop: 10, paddingTop: 10, borderTop: `1px solid ${FF.hair}`, fontFamily: FF_MONO, fontSize: 11, lineHeight: 1.55, color: FF.ink3, textWrap: 'pretty', ...style }}><span style={{ color: FF.teal, letterSpacing: '.12em', fontWeight: 600 }}>MOTION</span> {children}</div>; }
function Name({ children, size = 17, style }) { return <div style={{ fontSize: size, fontWeight: 700, letterSpacing: '-0.01em', color: FF.ink, lineHeight: 1.2, ...style }}>{children}</div>; }
function Desc({ children, size = 13.5, style }) { return <div style={{ fontSize: size, color: FF.ink2, lineHeight: 1.45, marginTop: 4, textWrap: 'pretty', ...style }}>{children}</div>; }
function Centered({ children, h }) { return <div style={{ display: 'flex', justifyContent: 'center', alignItems: 'center', height: h }}>{children}</div>; }
function Label({ children, color }) { return <FfMono style={{ fontSize: 10.5, color: color || FF.ink3, letterSpacing: '.14em', textTransform: 'uppercase' }}>{children}</FfMono>; }
