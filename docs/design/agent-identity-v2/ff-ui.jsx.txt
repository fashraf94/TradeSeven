function FfBtn({ kind = 'secondary', icon, children, onClick, disabled, full, small, ariaLabel, style, color, ariaPressed }) {

    accent: { background: ffA(FF.teal, 0.13), color: FF.teal, border: `1px solid ${ffA(FF.teal, 0.4)}` },
    secondary: { background: FF.raised, color: FF.ink, border: `1px solid ${FF.hair2}` },

// Dialog on desktop and tablet, bottom sheet on the phone. Escape closes, Tab
// cycles inside, focus returns to the opener.
function FfSheet({ title, eyebrow, onClose, children, phone, width = 560, label, footer }) {
  const ref = React.useRef(null);
  React.useEffect(() => {
    const prev = document.activeElement; const el = ref.current;
    const q = 'button:not([disabled]),[href],textarea,input,[tabindex]:not([tabindex="-1"])';
    const first = el && el.querySelector(q); if (first) first.focus();
    const onKey = (e) => {
      if (e.key === 'Escape') { e.stopPropagation(); onClose(); return; }
      if (e.key === 'Tab' && el) { const nodes = el.querySelectorAll(q); if (!nodes.length) return; const a = nodes[0], b = nodes[nodes.length - 1]; if (e.shiftKey && document.activeElement === a) { e.preventDefault(); b.focus(); } else if (!e.shiftKey && document.activeElement === b) { e.preventDefault(); a.focus(); } }
    };
    document.addEventListener('keydown', onKey);
    return () => { document.removeEventListener('keydown', onKey); if (prev && prev.focus) prev.focus(); };
  }, []);
  return (
    <div style={{ position: 'absolute', inset: 0, zIndex: 40, display: 'flex', alignItems: phone ? 'flex-end' : 'center', justifyContent: 'center', padding: phone ? 0 : 24 }}>
      <div className="ff-anim" onClick={onClose} style={{ position: 'absolute', inset: 0, background: 'rgba(5,6,9,0.72)', backdropFilter: 'blur(3px)', WebkitBackdropFilter: 'blur(3px)', animation: 'ffFadeIn .2s ease both' }} />
      <div ref={ref} role="dialog" aria-modal="true" aria-label={label || title} data-screen-label={label || title} className="ff-anim ff-scroll" style={{ position: 'relative', width: '100%', maxWidth: phone ? 'none' : width, maxHeight: phone ? '92%' : '90%', overflowY: 'auto', background: FF.bg, border: `1px solid ${FF.hair2}`, borderRadius: phone ? '22px 22px 0 0' : 20, padding: phone ? '12px 18px 28px' : '20px 24px 24px', boxShadow: '0 -20px 60px rgba(0,0,0,.5)', animation: phone ? 'ffSheetIn .3s cubic-bezier(.2,.8,.2,1) both' : 'ffDialogIn .2s ease both' }}>
        {phone && <div aria-hidden="true" style={{ width: 40, height: 4, borderRadius: 4, background: FF.hair2, margin: '0 auto 14px' }} />}
        <div style={{ display: 'flex', alignItems: 'flex-start', gap: 12 }}>
          <div style={{ flex: 1, minWidth: 0 }}>{eyebrow && <FfEyebrow color={FF.ink3}>{eyebrow}</FfEyebrow>}<h2 style={{ fontSize: 19, fontWeight: 700, letterSpacing: '-0.015em', margin: '4px 0 0', color: FF.ink, lineHeight: 1.3, textWrap: 'pretty' }}>{title}</h2></div>
          <button type="button" onClick={onClose} aria-label="Close" className="ff-reset ff-tap" style={{ cursor: 'pointer', width: 44, height: 44, borderRadius: 12, display: 'flex', alignItems: 'center', justifyContent: 'center', color: FF.ink2, background: FF.surface, border: `1px solid ${FF.hair}`, flexShrink: 0 }}><FfIco name="x" size={16} /></button>
        </div>
        <div style={{ marginTop: 16 }}>{children}</div>
        {footer && <div style={{ marginTop: 20, display: 'flex', gap: 10, flexWrap: 'wrap' }}>{footer}</div>}
      </div>
    </div>
  );
}

const FfRecCtx = React.createContext(false);
function RecNote({ children }) {
  const on = React.useContext(FfRecCtx);
  if (!on) return null;
  return <div style={{ fontFamily: FF_MONO, fontSize: 10, color: FF.gold, lineHeight: 1.5, padding: '6px 8px', borderRadius: 6, border: `1px dashed ${ffA(FF.gold, 0.4)}`, background: ffA(FF.gold, 0.05) }}>{children}</div>;
}
function Rec({ r, children, block }) {
  const on = React.useContext(FfRecCtx);
  if (!on) return children == null ? null : <>{children}</>;
  return <span style={{ position: 'relative', display: block ? 'block' : 'inline-block', outline: `1px dashed ${ffA(FF.gold, 0.5)}`, outlineOffset: 2, borderRadius: 3 }}>{children}<span style={{ position: 'absolute', top: -10, left: 2, zIndex: 5, fontFamily: FF_MONO, fontSize: 8, letterSpacing: '0.04em', color: FF.gold, background: FF.bg, padding: '0 4px', borderRadius: 3, whiteSpace: 'nowrap', lineHeight: '12px', border: `1px solid ${ffA(FF.gold, 0.3)}` }}>{r}</span></span>;
}

// Fixed design frame, scaled to fit the review pane.
function FfFrame({ w, h, min = 0.5, children }) {
  const ref = React.useRef(null); const [s, setS] = React.useState(min);
  React.useLayoutEffect(() => {
    const fit = () => { const el = ref.current; if (!el) return; const { width, height } = el.getBoundingClientRect(); if (width < 2 || height < 2) return; setS(Math.max(min, Math.min(1, (width - 24) / w, (height - 24) / h))); };
    fit(); const raf = requestAnimationFrame(fit); const ro = new ResizeObserver(fit); if (ref.current) ro.observe(ref.current); window.addEventListener('resize', fit);
    return () => { cancelAnimationFrame(raf); ro.disconnect(); window.removeEventListener('resize', fit); };
  }, [w, h, min]);
  return (
    <div ref={ref} className="ff-scroll" style={{ position: 'absolute', inset: 0, overflow: 'auto', zIndex: 2 }}>
      <div style={{ minWidth: '100%', minHeight: '100%', display: 'flex', alignItems: 'center', justifyContent: 'center', padding: 12 }}>
        <div style={{ width: w * s, height: h * s, flexShrink: 0 }}>
          <div style={{ width: w, height: h, transform: `scale(${s})`, transformOrigin: 'top left', borderRadius: 18 / s, overflow: 'hidden', border: `${1 / s}px solid ${FF.hair2}`, boxShadow: `0 ${30 / s}px ${90 / s}px rgba(0,0,0,.55)`, background: FF.bg, position: 'relative' }}>{children}</div>
        </div>
      </div>
    </div>
  );
}

function useFfMode(ref) {
  const [mode, setMode] = React.useState('desktop');
  React.useLayoutEffect(() => {
    const el = ref.current; if (!el) return;
    const upd = () => { const w = el.offsetWidth; setMode(w < 700 ? 'phone' : w < 1080 ? 'tablet' : 'desktop'); };
    upd(); const ro = new ResizeObserver(upd); ro.observe(el); return () => ro.disconnect();
  }, []);
  return mode;
}
