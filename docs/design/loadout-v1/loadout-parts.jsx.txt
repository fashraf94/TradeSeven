  return <span aria-hidden={label ? undefined : 'true'} style={{ width: s, height: s, borderRadius: Math.round(s / 4), background: FF.bg, border: `1px solid ${FF.hair}`, overflow: 'hidden', flexShrink: 0, display: 'block', ...style }}><Agent kind={k} core={1} armor={0} size={s} pal={pal} brain={!empty} face={face} label={label} /></span>;

function BookGlyph({ s = 48, dim }) { const c = dim ? FF.ink3 : FF.teal; return <span aria-hidden="true" style={{ width: s, height: s, borderRadius: Math.round(s / 4), background: FF.bg, border: `1px ${dim ? 'dashed' : 'solid'} ${dim ? // The loadout card: one game, its set, its book, its state, a door on each half.
function LoadoutCard({ game = 'BaggerBomb', kind = 'spec', book = BOOK.energy, state = 'next', running, w = 358, style, hot }) {
  const locked = state === 'locked', nobook = !book, tag = nobook ? 'nobook' : running ? 'running' : state;
  return <section aria-label={`${game} loadout`} data-screen-label={`${game} loadout card`} style={{ width: w, background: FF.surface, border: `1px solid ${hot ? ffA(FF.teal, .6) : locked ? FF.hair : ffA(FF.teal, .22)}`, boxShadow: hot ? `0 0 0 3px ${ffA(FF.teal, .12)}` : 'none', borderRadius: 18, padding: '12px 14px 12px', flexShrink: 0, ...style }}>
    <div style={{ display: 'flex', alignItems: 'center', justifyContent: 'space-between', gap: 8, minHeight: 24 }}><FfEyebrow color={locked ? FF.ink3 : FF.teal}>{game}</FfEyebrow><StateTag t={tag} /></div>
    <div style={{ display: 'flex', flexDirection: 'column', marginTop: 4 }}>
      <LRow icon={<Still k={kind} s={48} />} name={NAMES[kind]} sub={<><span>Set</span><span aria-hidden="true">·</span><RecordSlot /></>} locked={locked} tap={locked ? `${game} armor, locked. Open the armory to look.` : `Change ${game} armor`} />
      {book ? <LRow icon={<BookGlyph s={48} />} name={book.name} sub={<span>Book · {book.n} names</span>} locked={locked} tap={locked ? `${game} book, locked` : `Change ${game} book`} /> : <LRow icon={<BookGlyph s={48} dim />} name="No book" nameColor={FF.ink2} sub={<span style={{ color: FF.gold, fontWeight: 600 }}>Choose a list</span>} dashed tap="Choose a book" />}
    </div>
    {running && <RunningStrip kind={running} />}
    <StateLine state={state} nobook={nobook} running={running} />
  </section>;
}
// The one-line form: set small, book, state, and a door when there is one.
function LoadoutLine({ kind = 'contra', book = BOOK.ind, state = 'locked', eyebrow, door, dim, w, style }) {
  const sub = state === 'locked' ? ['lock', `Locked until the week ends · ${WEEK_END}`] : state === 'open' ? ['dot', 'Open · locks when you register'] : state === 'runningNow' ? ['dot', 'Ends at the close · keeps its armor'] : state === 'nobook' ? ['info', 'Deploy waits for a book'] : null;

  return <div role={hasDoor ? 'button' : undefined} tabIndex={hasDoor ? 0 : undefined} className={hasDoor ? 'pointer' : 'default', opacity: dim ? .75 : 1, ...style }}>
    <Still k={kind} s={40} />
    <div style={{ flex: 1, minWidth: 0 }}>
      {eyebrow && <FfMono style={{ display: 'block', fontSize: 10, letterSpacing: '.12em', textTransform: 'uppercase', color: FF.ink3, marginBottom: 3 }}>{eyebrow}</FfMono>}
      <div style={{ fontSize: 14, lineHeight: 1.3, color: FF.ink, textWrap: 'pretty' }}><span style={{ fontWeight: 700 }}>{NAMES[kind]}</span>{book ? <span style={{ color: FF.ink2 }}> · {book.name} · {book.n} names</span> : state === 'runningNow' ? null : <span style={{ color: FF.gold, fontWeight: 600 }}> · No book</span>}</div>
      {sub && <div style={{ display: 'flex', alignItems: 'center', gap: 5, marginTop: 3, color: state === 'locked' ? FF.ink3 : state === 'nobook' ? FF.gold : FF.teal }}><FfIco name={sub[0]} size={12} /><FfMono style={{ fontSize: 10, letterSpacing: '.08em', textTransform: 'uppercase', color: state === 'locked' ? FF.ink3 : state === 'nobook' ? FF.gold : FF.ink2 }}>{sub[1]}</FfMono></div>}
    </div>

  </div>;
