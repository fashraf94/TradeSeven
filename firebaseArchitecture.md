## Overview

### Goals
- Replace localStorage with Firebase for auth and battles
- Enable real-time sync across devices
- Maintain flexibility for UI/UX changes during beta
- Support gradual migration with feature flags
- Keep data structures extensible

### Not Migrating (Yet)
- Portfolio builder (stays in localStorage)
- Stats/leaderboards (Phase 2)
- Previous battles archive (Phase 2)

---

## Phase 1: Core Migration

### What We're Building
1. **Firebase Authentication** - Email/password + session persistence
2. **Firestore Collections** - Users, Battles, Challenges
3. **Real-time Sync** - Battle updates across devices
4. **Feature Flags** - Toggle Firebase on/off per feature

### Timeline
- Week 1: Setup + Auth migration
- Week 2: Battle system migration + testing
- Week 3: Real-time sync + feature flags
- Week 4: Full rollout + localStorage deprecation

## Firestore Collections

### Design Principles
1. **Versioning** - Every document has a `_v` field for schema version
2. **Nested Objects** - Group related data for easy extension
3. **Timestamps** - Track creation, updates, and important events
4. **Soft Deletes** - Use `archived` flag instead of deletion
5. **Flexible Fields** - Use `metadata` object for future additions

#### 3. Auth Service (`src/services/firebaseAuth.js`)
```javascript
import {
  createUserWithEmailAndPassword,
  signInWithEmailAndPassword,
  signOut,
  onAuthStateChanged
} from 'firebase/auth';
import { doc, setDoc, getDoc } from 'firebase/firestore';
import { auth, db } from '../config/firebase';

/**
 * Sign up new user
 */
export async function signUp(email, password, username) {
  try {
    // Create Firebase Auth user
    const userCredential = await createUserWithEmailAndPassword(auth, email, password);
    const user = userCredential.user;

    // Create Firestore user document
    await setDoc(doc(db, 'users', user.uid), {
      _v: 1,
      auth: {
        uid: user.uid,
        email: user.email,
        createdAt: new Date().toISOString(),
        lastLoginAt: new Date().toISOString()
      },
      profile: {
        username: username,
        displayName: username,
        avatarUrl: null,
        bio: null
      },
      stats: {
        xp: 0,
        level: 1,
        rank: "Beginner",
        wins: 0,
        losses: 0,
        totalBattles: 0,
        winStreak: 0,
        longestWinStreak: 0,
        totalXPEarned: 0
      },
      settings: {
        notifications: {
          battleStart: true,
          battleEnd: true,
          challengeAvailable: true
        },
        privacy: {
          showStats: true,
          allowChallenges: true
        }
      },
      achievements: [],
      metadata: {
        referralCode: null,
        premiumTier: null,
        flags: {}
      },
      archived: false,
      updatedAt: new Date().toISOString()
    });

    return { user, username };
  } catch (error) {
    console.error('Sign up error:', error);
    throw error;
  }
}

/**
 * Sign in existing user
 */
export async function signIn(email, password) {
  try {
    const userCredential = await signInWithEmailAndPassword(auth, email, password);
    const user = userCredential.user;

    // Update last login
    const userRef = doc(db, 'users', user.uid);
    await setDoc(userRef, {
      'auth.lastLoginAt': new Date().toISOString(),
      updatedAt: new Date().toISOString()
    }, { merge: true });

    // Fetch user data
    const userDoc = await getDoc(userRef);
    return { user, userData: userDoc.data() };
  } catch (error) {
    console.error('Sign in error:', error);
    throw error;
  }
}

/**
 * Sign out
 */
export async function logOut() {
  return signOut(auth);
}

/**
 * Listen to auth state changes
 */
export function onAuthChange(callback) {
  return onAuthStateChanged(auth, async (user) => {
    if (user) {
      const userDoc = await getDoc(doc(db, 'users', user.uid));
      callback({ user, userData: userDoc.data() });
    } else {
      callback(null);
    }
  });
}
```

## Migration Plan with Feature Flags

### Feature Flag System (`src/config/featureFlags.js`)

```javascript
/**
 * Feature Flags for gradual Firebase migration
 *
 * ROLLOUT PLAN:
 * Week 1: FIREBASE_AUTH = true (test with beta users)
 * Week 2: FIREBASE_BATTLES = true (test battle sync)
 * Week 3: FIREBASE_CHALLENGES = true (full feature set)
 * Week 4: Remove localStorage fallbacks
 */

export const FEATURE_FLAGS = {
  // Authentication
  FIREBASE_AUTH: false,           // true = Firebase Auth, false = localStorage

  // Battles
  FIREBASE_BATTLES: false,        // true = Firestore battles, false = localStorage

  // Challenges
  FIREBASE_CHALLENGES: false,     // true = Firestore challenges, false = localStorage

  // Real-time sync
  REALTIME_SYNC: false,           // true = Live updates, false = manual refresh

  // Debug mode
  DEBUG_MODE: true                // Log migration info to console
};

/**
 * Check if feature is enabled
 */
export function isFeatureEnabled(flagName) {
  return FEATURE_FLAGS[flagName] === true;
}

/**
 * Enable a feature flag (for testing)
 */
export function enableFeature(flagName) {
  FEATURE_FLAGS[flagName] = true;
  if (FEATURE_FLAGS.DEBUG_MODE) {
    console.log(`✅ Feature enabled: ${flagName}`);
  }
}

/**
 * Disable a feature flag (for rollback)
 */
export function disableFeature(flagName) {
  FEATURE_FLAGS[flagName] = false;
  if (FEATURE_FLAGS.DEBUG_MODE) {
    console.log(`❌ Feature disabled: ${flagName}`);
  }
}
```

### Hybrid Storage Service (`src/services/storageAdapter.js`)

```javascript
import { isFeatureEnabled } from '../config/featureFlags';
import * as firebaseAuth from './firebaseAuth';
import * as firebaseBattles from './firebaseBattles';
import { loadUser, saveUser } from './LocalStorage';

/**
 * Adaptive Auth Service
 * Routes to Firebase or localStorage based on feature flag
 */
export const authService = {
  async signUp(email, password, username) {
    if (isFeatureEnabled('FIREBASE_AUTH')) {
      return firebaseAuth.signUp(email, password, username);
    } else {
      // localStorage fallback
      const user = {
        username,
        xp: 0,
        level: 1,
        rank: "Beginner",
        wins: 0,
        losses: 0
      };
      saveUser(user);
      return { user };
    }
  },

  async signIn(email, password) {
    if (isFeatureEnabled('FIREBASE_AUTH')) {
      return firebaseAuth.signIn(email, password);
    } else {
      // localStorage fallback
      const user = loadUser();
      return { user };
    }
  },

  async signOut() {
    if (isFeatureEnabled('FIREBASE_AUTH')) {
      return firebaseAuth.logOut();
    } else {
      localStorage.removeItem('portfolioDuelUser');
    }
  }
};

/**
 * Adaptive Battle Service
 */
export const battleService = {
  async createBattle(battleData) {
    if (isFeatureEnabled('FIREBASE_BATTLES')) {
      return firebaseBattles.createBattle(battleData);
    } else {
      // localStorage fallback
      const battles = JSON.parse(localStorage.getItem('portfolioDuelBattles') || '[]');
      battles.push(battleData);
      localStorage.setItem('portfolioDuelBattles', JSON.stringify(battles));
      return battleData;
    }
  },

  async getBattles(userId) {
    if (isFeatureEnabled('FIREBASE_BATTLES')) {
      return firebaseBattles.getUserBattles(userId);
    } else {
      return JSON.parse(localStorage.getItem('portfolioDuelBattles') || '[]');
    }
  }
};
```

## Security Rules

### Firestore Security Rules (`firestore.rules`)

```javascript
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Helper functions
    function isSignedIn() {
      return request.auth != null;
    }

    function isOwner(uid) {
      return request.auth.uid == uid;
    }

    function isValidUsername(username) {
      return username is string
        && username.size() >= 3
        && username.size() <= 20;
    }

    // Users collection
    match /users/{userId} {
      // Anyone can read public user profiles (for leaderboards)
      allow read: if isSignedIn();

      // Users can only create/update their own profile
      allow create: if isSignedIn()
        && isOwner(userId)
        && isValidUsername(request.resource.data.profile.username);

      allow update: if isSignedIn() && isOwner(userId);

      // No deletes (soft delete with archived flag)
      allow delete: if false;
    }

    // Battles collection
    match /battles/{battleId} {
      // Anyone can read battles (for browsing/joining)
      allow read: if isSignedIn();

      // Only authenticated users can create battles
      allow create: if isSignedIn()
        && request.resource.data.creator.uid == request.auth.uid;

      // Only participants can update battles
      allow update: if isSignedIn()
        && (resource.data.creator.uid == request.auth.uid
           || resource.data.opponent.uid == request.auth.uid);

      // No deletes (soft delete with archived flag)
      allow delete: if false;
    }

    // Challenges collection
    match /challenges/{challengeId} {
      // Only battle participants can read challenges
      allow read: if isSignedIn();

      // Only the player can create/update their challenges
      allow create: if isSignedIn()
        && request.resource.data.player.uid == request.auth.uid;

      allow update: if isSignedIn()
        && resource.data.player.uid == request.auth.uid;

      // No deletes
      allow delete: if false;
    }
  }
}
```

## Real-time Sync Strategy

### Battle Sync Service (`src/services/firebaseBattles.js`)

```javascript
import {
  collection,
  doc,
  addDoc,
  updateDoc,
  query,
  where,
  onSnapshot,
  getDocs
} from 'firebase/firestore';
import { db } from '../config/firebase';

/**
 * Create a new battle
 */
export async function createBattle(battleData) {
  try {
    const battleRef = await addDoc(collection(db, 'battles'), {
      _v: 1,
      challengeCode: battleData.challengeCode,

      creator: {
        uid: battleData.creator.uid,
        username: battleData.creator.username,
        portfolioName: battleData.portfolioName,
        portfolio: battleData.creatorPortfolio,
        portfolioType: battleData.portfolioType
      },

      opponent: {
        uid: null,
        username: null,
        portfolioName: null,
        portfolio: null,
        portfolioType: null
      },

      timeline: {
        createdAt: new Date().toISOString(),
        startDate: null,
        endDate: null,
        completedAt: null
      },

      state: {
        status: 'waiting',
        currentDay: 0,
        startingPrices: null,
        endingPrices: null
      },

      result: null,
      challengeIds: [],
      metadata: {
        spectatorCount: 0,
        featured: false,
        tags: []
      },

      archived: false,
      updatedAt: new Date().toISOString()
    });

    return { id: battleRef.id, ...battleData };
  } catch (error) {
    console.error('Error creating battle:', error);
    throw error;
  }
}

/**
 * Join a battle by challenge code
 */
export async function joinBattle(challengeCode, opponentData) {
  try {
    // Find battle by challenge code
    const q = query(
      collection(db, 'battles'),
      where('challengeCode', '==', challengeCode),
      where('state.status', '==', 'waiting')
    );

    const snapshot = await getDocs(q);

    if (snapshot.empty) {
      throw new Error('Battle not found or already started');
    }

    const battleDoc = snapshot.docs[0];
    const battleRef = doc(db, 'battles', battleDoc.id);

    // Calculate start/end times
    const now = new Date();
    const startDate = now.toISOString();
    const endDate = new Date(now.getTime() + (24 * 60 * 60 * 1000)).toISOString(); // 24 hours

    // Update battle with opponent
    await updateDoc(battleRef, {
      'opponent.uid': opponentData.uid,
      'opponent.username': opponentData.username,
      'opponent.portfolioName': opponentData.portfolioName,
      'opponent.portfolio': opponentData.portfolio,
      'opponent.portfolioType': opponentData.portfolioType,

      'timeline.startDate': startDate,
      'timeline.endDate': endDate,

      'state.status': 'active',
      'state.currentDay': 1,
      'state.startingPrices': opponentData.startingPrices,

      updatedAt: new Date().toISOString()
    });

    return { id: battleDoc.id, ...battleDoc.data() };
  } catch (error) {
    console.error('Error joining battle:', error);
    throw error;
  }
}

/**
 * Get all battles for a user
 */
export async function getUserBattles(userId) {
  try {
    const q1 = query(
      collection(db, 'battles'),
      where('creator.uid', '==', userId),
      where('archived', '==', false)
    );

    const q2 = query(
      collection(db, 'battles'),
      where('opponent.uid', '==', userId),
      where('archived', '==', false)
    );

    const [snapshot1, snapshot2] = await Promise.all([
      getDocs(q1),
      getDocs(q2)
    ]);

    const battles = [
      ...snapshot1.docs.map(doc => ({ id: doc.id, ...doc.data() })),
      ...snapshot2.docs.map(doc => ({ id: doc.id, ...doc.data() }))
    ];

    return battles;
  } catch (error) {
    console.error('Error fetching battles:', error);
    throw error;
  }
}

/**
 * Subscribe to real-time battle updates
 */
export function subscribeToBattles(userId, callback) {
  const q1 = query(
    collection(db, 'battles'),
    where('creator.uid', '==', userId),
    where('archived', '==', false)
  );

  const q2 = query(
    collection(db, 'battles'),
    where('opponent.uid', '==', userId),
    where('archived', '==', false)
  );

  // Listen to creator battles
  const unsubscribe1 = onSnapshot(q1, (snapshot) => {
    const battles = snapshot.docs.map(doc => ({ id: doc.id, ...doc.data() }));
    callback(battles, 'creator');
  });

  // Listen to opponent battles
  const unsubscribe2 = onSnapshot(q2, (snapshot) => {
    const battles = snapshot.docs.map(doc => ({ id: doc.id, ...doc.data() }));
    callback(battles, 'opponent');
  });

  // Return combined unsubscribe function
  return () => {
    unsubscribe1();
    unsubscribe2();
  };
}

/**
 * Complete a battle
 */
export async function completeBattle(battleId, resultData) {
  try {
    const battleRef = doc(db, 'battles', battleId);

    await updateDoc(battleRef, {
      'state.status': 'completed',
      'state.endingPrices': resultData.endingPrices,
      'timeline.completedAt': new Date().toISOString(),
      result: resultData.result,
      updatedAt: new Date().toISOString()
    });

    return true;
  } catch (error) {
    console.error('Error completing battle:', error);
    throw error;
  }
}
```

```javascript
import { isFeatureEnabled } from './config/featureFlags';

function App() {

    } else {
      // localStorage fallback
      const savedUser = loadUser();
      setUser(savedUser);
    }
  }, []);

  useEffect(() => {
    // Real-time battle sync
    if (user && isFeatureEnabled('REALTIME_SYNC')) {
      const unsubscribe = subscribeToBattles(user.auth.uid, (newBattles) => {
        setBattles(prevBattles => {
          // Merge new battles with existing
          const merged = [...prevBattles];
          newBattles.forEach(newBattle => {
            const index = merged.findIndex(b => b.id === newBattle.id);
            if (index >= 0) {
              merged[index] = newBattle;
            } else {
              merged.push(newBattle);
            }
          });
          return merged;
        });
      });

      return unsubscribe;
    } else if (user) {
      // Manual fetch
      battleService.getBattles(user.auth.uid).then(setBattles);
    }
  }, [user]);

  // ... rest of your app
}
```

```javascript
import { useState } from 'react';
import { authService } from '../services/storageAdapter';

function SignUp() {
  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');
  const [username, setUsername] = useState('');

      // User will be updated via onAuthChange listener
    } catch (err) {
      setError(err.message);
    }
  };

  return (
    <form onSubmit={handleSignUp}>
      <input
        type="text"
        placeholder="Username"
        value={username}
        onChange={(e) => setUsername(e.target.value)}
      />
      <input
        type="email"
        placeholder="Email"
        value={email}
        onChange={(e) => setEmail(e.target.value)}
      />
      <input
        type="password"
        placeholder="Password"
        value={password}
        onChange={(e) => setPassword(e.target.value)}
      />
      <button type="submit">Sign Up</button>
      {error && <p style={{ color: 'red' }}>{error}</p>}
    </form>
  );
```
